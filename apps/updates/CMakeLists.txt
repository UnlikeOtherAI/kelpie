cmake_minimum_required(VERSION 3.16)
project(kelpie_update_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
include(CTest)
include(FetchContent)
find_package(nlohmann_json CONFIG QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
  FetchContent_Declare(nlohmann_json
    URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3
    URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(nlohmann_json)
endif()
add_executable(release_updates_test tests/release_updates_test.cpp)
target_link_libraries(release_updates_test PRIVATE nlohmann_json::nlohmann_json)
if(WIN32)
  target_compile_definitions(release_updates_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(release_updates_test PRIVATE /UNDEBUG)
else()
  target_compile_options(release_updates_test PRIVATE -UNDEBUG)
endif()
add_test(NAME release_updates_test COMMAND release_updates_test)
