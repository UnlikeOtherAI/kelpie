add_library(kelpie_desktop_account STATIC
  account_protocol.cpp account_login.cpp account_service.cpp)
target_compile_features(kelpie_desktop_account PUBLIC cxx_std_20)
target_include_directories(kelpie_desktop_account PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(kelpie_desktop_account PUBLIC kelpie_core_state kelpie_core_protocol kelpie_core_automation
  nlohmann_json::nlohmann_json httplib::httplib)
if(WIN32)
  target_sources(kelpie_desktop_account PRIVATE account_transport_windows.cpp)
  target_compile_definitions(kelpie_desktop_account PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(kelpie_desktop_account PRIVATE winhttp bcrypt crypt32 ws2_32)
else()
  find_package(OpenSSL REQUIRED)
  find_package(CURL REQUIRED)
  target_sources(kelpie_desktop_account PRIVATE account_transport_curl.cpp)
  target_link_libraries(kelpie_desktop_account PRIVATE OpenSSL::Crypto CURL::libcurl)
endif()
if(BUILD_TESTING)
  add_executable(account_service_test tests/account_service_test.cpp)
  target_link_libraries(account_service_test PRIVATE kelpie_desktop_account)
  if(MSVC)
    target_compile_options(account_service_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(account_service_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME account_service_test COMMAND account_service_test)
endif()
