add_library(kelpie_desktop_connect STATIC connect_client.cpp connect_executor.cpp)
target_compile_features(kelpie_desktop_connect PUBLIC cxx_std_20)
target_include_directories(kelpie_desktop_connect PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(kelpie_desktop_connect PUBLIC kelpie_desktop_account kelpie_core_connect kelpie_engine_chromium_desktop)
if(WIN32)
  target_sources(kelpie_desktop_connect PRIVATE connect_transport_windows.cpp)
  target_compile_definitions(kelpie_desktop_connect PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(kelpie_desktop_connect PRIVATE winhttp)
else()
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  target_sources(kelpie_desktop_connect PRIVATE connect_transport_curl.cpp)
  target_link_libraries(kelpie_desktop_connect PRIVATE CURL::libcurl OpenSSL::Crypto)
endif()
if(BUILD_TESTING)
  add_executable(connect_executor_test tests/connect_executor_test.cpp)
  target_link_libraries(connect_executor_test PRIVATE kelpie_desktop_connect)
  kelpie_enable_test_assertions(connect_executor_test)
  add_test(NAME connect_executor_test COMMAND connect_executor_test)
endif()
